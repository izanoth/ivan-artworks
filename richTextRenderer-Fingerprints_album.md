import PostPreview from '@/blog/components/PostPreview';

interface Post {
    id: string;
    title: string;
    content: string;
    category: {
		  id: number;
		  name: string;    
    } | null;
    author: {
        name: string;
        email: string;
    } | null;
    source: string;
    createdAt: Date;
    updatedAt: Date;
    published: boolean;
}


export default function Blog() {
    const [posts, setPosts] = useState<Post[]>([]);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        const fetchPosts = async () => {
            try {
                const res = await fetch('/api/blog');
                if (res.ok) {
                    const data: Post[] = await res.json();
                    const sortedPosts = data.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
                    setPosts(sortedPosts);
                }
            } catch (error) {
                console.error('Error fetching guests:', error);
            } finally {
                setLoading(false);
            }
        };
        fetchPosts();

    if (loading) {
        return (
            <div className="flex flex-col items-center justify-center min-h-screen">
                <p className="blinking-text block text-xs pb-4">LOADING</p>
                <span className="loader"></span>
            </div>
        );
    }

    return (
        <section>
            <main className="container">
                {posts.map((post) => (
                    <div key={post.id} className="post">
                        <PostPreview post={post}/>
                    </div>
                ))}
            </main>
        </section>
    );
};

## MODEL ✨

A melhor abordagem para permitir itálicos nos títulos dos posts, sem usar `ReactMarkdown` e de forma segura, é criar um componente utilitário que interprete uma sintaxe simples (como `*texto*` para itálico) e renderize o texto como elementos React, usando `<em>` para as partes em itálico.

Minha estratégia será:

1.  **Criar um componente `RichTextRenderer.tsx`:** Este componente receberá o título como uma string, fará o parsing para identificar as seções em itálico (marcadas com asteriscos, por exemplo, `*texto*`), e renderizará essas seções dentro de tags `<em>`. Isso evita `dangerouslySetInnerHTML` e os riscos de XSS.
2.  **Atualizar `PostDetail.tsx`:** Substituir a renderização direta de `post.title` pelo novo componente `RichTextRenderer`.
3.  **Atualizar `PostPreview.tsx`:** Substituir a renderização direta de `post.title` pelo novo componente `RichTextRenderer`.
4.  **Ajustar `generateMetadata`:** O título para os metadados (SEO) geralmente precisa ser texto puro. Adicionarei uma função para remover as marcas de itálico do título antes de usá-lo em `generateMetadata` em `src/app/(alternatives)/blog/p/[id]/page.tsx`.
